# Minimal dependency set for the Vercel deployment of the mapping editor.
#
# Vercel installs ONLY this (via `installCommand` in vercel.json), NOT the bots'
# full pyproject/requirements — the editor's request path imports just these plus
# the Python stdlib. Do NOT add Pyrogram / SQLAlchemy / Pillow / etc. here: the
# whole point is a tiny, fast serverless bundle. The heavy modules that would
# need them are never imported on Vercel (their `Container` type-import is
# TYPE_CHECKING-guarded).
fastapi>=0.117.0
redis>=5.0.0
structlog>=24.1.0
# Required, not optional: `from fastapi import Request` imports starlette.formparsers,
# which does an UNGUARDED `import python_multipart`. pip does NOT auto-install it
# (it's a fastapi extra), so without this the function crashes on every request
# with FUNCTION_INVOCATION_FAILED. (ujson/orjson ARE optional — guarded by fastapi.)
python-multipart>=0.0.9
